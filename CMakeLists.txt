cmake_minimum_required(VERSION 3.10)
project(fem LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(fem_solver src/electrostatic_analyzer.cpp)
target_include_directories(fem_solver PUBLIC src)
add_executable(fem src/main.cpp)
target_link_libraries(fem PRIVATE fem_solver)
set_target_properties(fem PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(fem PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin")
endforeach()

include(CTest)
if(BUILD_TESTING)
    foreach(name physics circuit)
        add_executable(${name}_regression tests/${name}_regression.cpp)
        target_link_libraries(${name}_regression PRIVATE fem_solver)
        file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-output/${name}")
        add_test(NAME ${name}_regression COMMAND ${name}_regression "${CMAKE_BINARY_DIR}/test-output/${name}")
    endforeach()
endif()
